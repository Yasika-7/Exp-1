const express = require("express"); const cors = require("cors"); const app = express(); app.use(cors()); app.use(express.json()); // Frontend app.get("/", (req, res) => { res.send(` My Web Application
`); }); // Backend API app.get("/api", (req, res) => { res.json({ message: "Backend Connected Successfully!" }); }); // Start Server const PORT = process.env.PORT || 5000; app.listen(PORT, () => { console.log("Server running on port " + PORT); });
