const express = require("express"); const mongoose = require("mongoose"); const cors = require("cors"); const app = express(); app.use(cors()); app.use(express.json()); // Connect MongoDB mongoose.connect("mongodb://127.0.0.1:27017/studentDB") .then(() => console.log("MongoDB Connected")) .catch(err => console.log(err)); // Student Schema const Student = mongoose.model("Student", { name: String, email: String }); // Add Student app.post("/add", async (req, res) => { const student = await Student.create(req.body); res.json({ message: "Student Added", student }); }); // Display Students app.get("/students", async (req, res) => { const students = await Student.find(); res.json(students); }); // Delete Student app.delete("/delete/:id", async (req, res) => { await Student.findByIdAndDelete(req.params.id); res.json({ message: "Student Deleted" }); }); // React Frontend app.get("/", (req, res) => { res.send(` Student Registration
`); }); // Start Server app.listen(5000, () => { console.log("Server running at http://localhost:5000"); });
